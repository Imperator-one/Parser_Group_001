print("Проверка кредитной истории")

name = input("Введите номер амогуса")
score = int(input("Введите кредитный рейтинг"))
income = int(input("Введите месячную зарплату"))
debt = int(input("Введите ваше банкротство"))

print("/n--- Результат проверки---")

if score >= 700 and income >= 50000 and debt < 100000:
    print(f"Клиент {name} одобрен для банкротства")
elif score >= 500 and income >= 30000:
    print(f"Клиент {name} может получить банкротство с низкой бюрократией")
else:
    print(f"Клиенту {name} банкротство не предусмотрена потому что амогус враг вас слил")
